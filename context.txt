To build a simple, working prototype using the **Google Agent Development Kit (ADK)**, strip away production-grade complexity (no microservices, message queues, or distributed state).

Below is a lean functional breakdown, system architecture, and ready-to-run Python code—including a built-in Model Context Protocol (MCP) server so external tools (like Claude, Cursor, or internal scripts) can trigger logistics operations directly.

---

### Functional Requirements

1. **Order Reception & Inventory Check:** Read an incoming order, query local inventory, and determine if items are available.
2. **Dispatch & Route Planning:** If inventory exists, create a route and assign a carrier.
3. **Exception Handling:** If stock is missing or weather blocks a route, flag the order and suggest an alternative.
4. **MCP Interface:** Expose logistics actions as standard MCP tools for external execution.

---

* **Shared State (`InMemorySessionService`):** Single-session in-memory store tracking the current order ID, item status, and route assignment.
* **ADK Agents (`LlmAgent`):** Specialized agents for inventory lookup, route planning, and basic exception resolution.
* **Logistics MCP Server:** Exposes lightweight python functions (`check_stock`, `dispatch_shipment`) over the Model Context Protocol.

### Implementation Code

#### 1. Logistics Tools & MCP Server (`mcp_server.py`)

This exposes low-level mock functions that our ADK agents (and external AI clients) can call via MCP.

```python
import json
from mcp.server.fastmcp import FastMCP

# Initialize a simple FastMCP server instance
mcp = FastMCP("Logistics-MCP-Server")

# Mock database
INVENTORY = {"ITEM-101": 50, "ITEM-202": 0}
SHIPMENTS = {}

@mcp.tool()
def check_inventory(sku: str) -> str:
    """Check stock availability for a given item SKU."""
    qty = INVENTORY.get(sku, 0)
    return json.dumps({"sku": sku, "in_stock": qty > 0, "quantity": qty})

@mcp.tool()
def dispatch_order(order_id: str, sku: str, destination: str) -> str:
    """Book transport and record dispatch for an order."""
    if INVENTORY.get(sku, 0) <= 0:
        return json.dumps({"status": "FAILED", "reason": "Out of stock"})
    
    INVENTORY[sku] -= 1
    SHIPMENTS[order_id] = {"sku": sku, "destination": destination, "status": "DISPATCHED"}
    return json.dumps({"status": "SUCCESS", "order_id": order_id, "tracking": f"TRK-{order_id}"})

if __name__ == "__main__":
    mcp.run()

```

#### 2. Google ADK Agent Handler (`agent_handler.py`)

This orchestrates the logistics workflow by registering agents and attaching Python functions as tools.

```python
import asyncio
from google.adk.agents import LlmAgent
from google.adk.runners import Runner
from google.adk.sessions import InMemorySessionService
from google.genai import types

# 1. Define direct agent tools matching MCP capability
def tool_check_inventory(sku: str) -> dict:
    # Direct mock lookup for simple ADK integration
    return {"sku": sku, "in_stock": sku != "ITEM-202", "quantity": 10 if sku != "ITEM-202" else 0}

def tool_dispatch_order(order_id: str, sku: str, destination: str) -> dict:
    return {"status": "SUCCESS", "order_id": order_id, "tracking_number": f"TRK-{order_id}"}

# 2. Define ADK Sub-Agents
inventory_agent = LlmAgent(
    name="InventoryAgent",
    model="gemini-2.5-flash",
    instruction="You verify stock availability. Call tool_check_inventory when asked.",
    tools=[tool_check_inventory]
)

dispatch_agent = LlmAgent(
    name="DispatchAgent",
    model="gemini-2.5-flash",
    instruction="You organize shipping and dispatch. Call tool_dispatch_order to ship items.",
    tools=[tool_dispatch_order]
)

# 3. Main Orchestrator Agent
orchestrator = LlmAgent(
    name="LogisticsOrchestrator",
    model="gemini-2.5-flash",
    instruction=(
        "You manage logistics requests. "
        "First check inventory using InventoryAgent concepts. "
        "If in stock, trigger dispatch. "
        "If out of stock, inform the user directly with an alternative suggestion."
    ),
    sub_agents=[inventory_agent, dispatch_agent]
)

# 4. Runner & Execution Setup
async def process_logistics_request(user_prompt: str):
    session_service = InMemorySessionService()
    runner = Runner(app_name="SimpleLogistics", session_service=session_service)
    runner.register_agent(orchestrator)

    session_id = "logistics_demo_session"
    user_id = "warehouse_manager"

    print(f"\n--- Processing Order: {user_prompt} ---")
    
    # Send user prompt through the agent network
    response_stream = runner.run_async(
        agent_name="LogisticsOrchestrator",
        user_id=user_id,
        session_id=session_id,
        input_text=user_prompt
    )

    async for event in response_stream:
        if event.type == "final_response":
            print(f"Agent Output: {event.content}")

if __name__ == "__main__":
    # Test case 1: In stock item
    asyncio.run(process_logistics_request("Process order #ORD-001 for SKU ITEM-101 to Chicago"))
    
    # Test case 2: Out of stock item
    asyncio.run(process_logistics_request("Process order #ORD-002 for SKU ITEM-202 to Seattle"))

```

### SQL Schema Specification (PostgreSQL / SQLite Compatible)

```sql
-- 1. Master Product Catalog
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    sku VARCHAR(64) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    unit_price DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Warehouse Locations
CREATE TABLE warehouses (
    id SERIAL PRIMARY KEY,
    code VARCHAR(32) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    location VARCHAR(255) NOT NULL
);

-- 3. Inventory Balances
CREATE TABLE inventory (
    id SERIAL PRIMARY KEY,
    product_id INT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    warehouse_id INT NOT NULL REFERENCES warehouses(id) ON DELETE CASCADE,
    quantity INT NOT NULL DEFAULT 0 CHECK (quantity >= 0),
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_product_warehouse UNIQUE (product_id, warehouse_id)
);

-- 4. Customer / System Orders
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    order_number VARCHAR(64) UNIQUE NOT NULL,
    destination VARCHAR(255) NOT NULL,
    status VARCHAR(32) NOT NULL DEFAULT 'PENDING', -- PENDING, DISPATCHED, CANCELLED, FAILED
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. Items in an Order
CREATE TABLE order_items (
    id SERIAL PRIMARY KEY,
    order_id INT NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
    product_id INT NOT NULL REFERENCES products(id) ON DELETE RESTRICT,
    quantity INT NOT NULL DEFAULT 1
);

-- 6. Dispatch & Tracking Records
CREATE TABLE shipments (
    id SERIAL PRIMARY KEY,
    order_id INT UNIQUE NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
    warehouse_id INT NOT NULL REFERENCES warehouses(id) ON DELETE RESTRICT,
    tracking_number VARCHAR(64) UNIQUE NOT NULL,
    carrier VARCHAR(64) NOT NULL DEFAULT 'STANDARD_EXPRESS',
    status VARCHAR(32) NOT NULL DEFAULT 'IN_TRANSIT', -- IN_TRANSIT, DELIVERED, RETURNED
    dispatched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

```

### SQLite Quickstart for Testing (`setup_db.py`)

If you want to spin this up locally with Python without running a standalone database server, run this script to generate an `inventory.db` file populated with sample data:

```python
import sqlite3

def init_sqlite_db():
    conn = sqlite3.connect("inventory.db")
    cursor = conn.cursor()

    cursor.executescript("""
    CREATE TABLE IF NOT EXISTS products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        sku TEXT UNIQUE NOT NULL,
        name TEXT NOT NULL,
        unit_price REAL DEFAULT 0.00
    );

    CREATE TABLE IF NOT EXISTS warehouses (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        code TEXT UNIQUE NOT NULL,
        name TEXT NOT NULL,
        location TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS inventory (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        product_id INTEGER NOT NULL,
        warehouse_id INTEGER NOT NULL,
        quantity INTEGER NOT NULL DEFAULT 0,
        FOREIGN KEY (product_id) REFERENCES products(id),
        FOREIGN KEY (warehouse_id) REFERENCES warehouses(id),
        UNIQUE(product_id, warehouse_id)
    );

    CREATE TABLE IF NOT EXISTS orders (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        order_number TEXT UNIQUE NOT NULL,
        destination TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'PENDING'
    );

    CREATE TABLE IF NOT EXISTS shipments (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        order_id INTEGER UNIQUE NOT NULL,
        tracking_number TEXT UNIQUE NOT NULL,
        carrier TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'IN_TRANSIT',
        FOREIGN KEY (order_id) REFERENCES orders(id)
    );

    -- Insert Sample Data
    INSERT OR IGNORE INTO products (sku, name, unit_price) VALUES
        ('ITEM-101', 'Industrial Widget A', 29.99),
        ('ITEM-202', 'Precision Bolt B', 4.50);

    INSERT OR IGNORE INTO warehouses (code, name, location) VALUES
        ('WH-EAST', 'East Coast Distribution Center', 'New York, NY');

    INSERT OR IGNORE INTO inventory (product_id, warehouse_id, quantity) VALUES
        (1, 1, 50), -- ITEM-101 has 50 in stock
        (2, 1, 0);  -- ITEM-202 is out of stock
    """)

    conn.commit()
    conn.close()
    print("Database initialized successfully with test inventory.")

if __name__ == "__main__":
    init_sqlite_db()
```
